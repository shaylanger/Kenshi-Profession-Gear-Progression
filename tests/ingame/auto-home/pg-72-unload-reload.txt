# covers: PG 106 (an NPC streams out and back in: his item does not reroll)

# design: a spawned Drifter (own squad, not the player's) gets a rolled hat, is unloaded and reloaded right away (the game also
#   reloads squads near the player by itself: "(the game had already reloaded him)" is fine).
# verify: same key + affixes after reload; ProfessionGear.log shows `restore item id=<K>` (restored, not a new roll)
@wait-world
@sleep 8
speed 0
@set N spawn "Drifter" Drifters near Shay dist 45 count 1 ~ spawned 1/1 [^:]+: .+? (#\d+(?:/\d+)?)
setname ${N} "PG Stream" ~ -> PG Stream
give "PG Stream" 2185-gamedata.base 1
@set K pg_info "PG Stream" "Rattan Hat" ~ Rattan Hat key=(pgp\S+)
@set A pg_info "PG Stream" "Rattan Hat" ~ affixes=(.*)$
unload "PG Stream"
reload "PG Stream" ~ PG Stream
@sleep 5
where "PG Stream" ~ PG Stream
pg_info "PG Stream" "Rattan Hat" ~ Rattan Hat key=${K} equipped=[01] tier=\d+ affixes=${A}$
@log D:\Steam\steamapps\common\Kenshi\mods\ProfessionGearProgression\ProfessionGear.log ~ restore item id=${K}
